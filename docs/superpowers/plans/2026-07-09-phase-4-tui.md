```go
// Package pubsub is an in-process typed event broker (F19). Subscribers
// get bounded per-subscriber buffers; a slow subscriber drops non-terminal
// events rather than blocking publishers, but terminal events (turn end,
// approval requested) block the publisher until delivered or the
// subscriber's context is cancelled (crush's "must-deliver" rule).
//
// This is a clean-room implementation of the public behavior described in
// docs/12 F19; it does not derive from crush's FSL-licensed source.
package pubsub

import (
	"context"
	"sync"
	"time"
)

// Option configures a subscription.
type Option[T any] func(*subscription[T])

// WithBufferSize sets the per-subscriber buffer for non-terminal events.
// Default is 16. A full buffer drops the oldest non-terminal event and
// continues (drop-head, never block the publisher).
func WithBufferSize[T any](n int) Option[T] {
	return func(s *subscription[T]) {
		if n >= 0 {
			s.buffer = n
		}
	}
}

// WithTerminal marks the subscriber as a terminal-event receiver. Terminal
// events (Type agreed out-of-band; the broker does not decide which types
// are terminal — the publisher uses PublishTerminal for them) block the
// publisher on a per-subscriber basis until that subscriber delivers or its
// context is cancelled.
func WithTerminal[T any]() Option[T] {
	return func(s *subscription[T]) {
		s.terminal = true
	}
}

type subscription[T any] struct {
	ch       chan Event[T]
	buffer   int
	terminal bool
}

// Broker is a generic typed pub/sub broker. The zero value is not usable;
// use NewBroker.
type Broker[T any] struct {
	mu     sync.RWMutex
	subs   []*subscription[T]
	closed bool
}

// NewBroker creates a broker. Subscribe before publishing.
func NewBroker[T any]() *Broker[T] {
	return &Broker[T]{}
}

// Subscribe returns a receive channel. When ctx is cancelled the channel is
// closed and the subscription is removed. A terminal subscriber also
// unblocks any pending PublishTerminal on ctx cancellation.
func (b *Broker[T]) Subscribe(ctx context.Context, opts ...Option[T]) <-chan Event[T] {
	s := &subscription[T]{buffer: 16}
	for _, opt := range opts {
		opt(s)
	}
	s.ch = make(chan Event[T], s.buffer)

	b.mu.Lock()
	b.subs = append(b.subs, s)
	b.mu.Unlock()

	go func() {
		<-ctx.Done()
		b.mu.Lock()
		for i, sub := range b.subs {
			if sub == s {
				b.subs = append(b.subs[:i], b.subs[i+1:]...)
				break
			}
		}
		b.mu.Unlock()
		// Unblock any pending terminal publish waiting on this subscriber.
		if s.terminal {
			// Closing the channel is the signal: a blocked send in
			// PublishTerminal selects ctx.Done(); we additionally drain
			// nothing here — the blocked send will return via ctx.Done.
		}
		close(s.ch)
	}()
	return s.ch
}

// Publish broadcasts a non-terminal event. A full subscriber buffer drops
// the oldest event (drop-head) and continues; the publisher never blocks.
func (b *Broker[T]) Publish(typ string, payload T) {
	b.publish(typ, payload, false)
}

// PublishTerminal broadcasts a terminal event. For terminal subscribers
// (WithTerminal) the call blocks until the event is delivered or the
// subscriber's context is cancelled. Non-terminal subscribers receive it
// like a normal event (best-effort).
func (b *Broker[T]) PublishTerminal(typ string, payload T) {
	b.publish(typ, payload, true)
}

func (b *Broker[T]) publish(typ string, payload T, terminal bool) {
	ev := Event[T]{Type: typ, Payload: payload, PublishedAt: time.Now()}
	b.mu.RLock()
	subs := append([]*subscription[T](nil), b.subs...)
	b.mu.RUnlock()

	for _, s := range subs {
		if terminal && s.terminal {
			// Must deliver: block until the subscriber reads or its ctx
			// cancels (which closes s.ch). A closed channel send panics,
			// so recover.
			sendBlocking(s.ch, ev)
			continue
		}
		// Best-effort: non-blocking send; drop-head on full buffer.
		select {
		case s.ch <- ev:
		default:
			dropHead(s.ch)
			select {
			case s.ch <- ev:
			default:
			}
		}
	}
}

func sendBlocking[T any](ch chan Event[T], ev Event[T]) {
	defer func() { recover() }() // closed-channel send during cancel
	ch <- ev
}

func dropHead[T any](ch chan Event[T]) {
	select {
	case <-ch:
	default:
	}
}

// Close shuts down the broker: all subscriber channels are closed and
// subsequent Publish calls are no-ops.
func (b *Broker[T]) Close() {
	b.mu.Lock()
	defer b.mu.Unlock()
	if b.closed {
		return
	}
	b.closed = true
	for _, s := range b.subs {
		close(s.ch)
	}
	b.subs = nil
}
```

```go
// Package csync provides small generic concurrent-state primitives (F19 R3)
// adopted for new concurrent state instead of ad-hoc mutex fields on
// session.State. They are thin wrappers over atomic/sync with copy-on-read
// semantics so callers never observe a mutated value mid-flight.
package csync

import "sync"

// Value is a guarded single value. Load returns (zero, false) until the
// first Store, (value, true) afterwards.
type Value[T any] struct {
	mu   sync.RWMutex
	v    T
	set  bool
}

func (v *Value[T]) Store(x T) {
	v.mu.Lock()
	v.v = x
	v.set = true
	v.mu.Unlock()
}

func (v *Value[T]) Load() (T, bool) {
	v.mu.RLock()
	defer v.mu.RUnlock()
	return v.v, v.set
}
```

```go
package csync

import "sync"

// Slice is a guarded append-only slice. Load returns a fresh copy.
type Slice[T any] struct {
	mu sync.Mutex
	s  []T
}

func (s *Slice[T]) Append(x T) {
	s.mu.Lock()
	s.s = append(s.s, x)
	s.mu.Unlock()
}

func (s *Slice[T]) Load() []T {
	s.mu.Lock()
	defer s.mu.Unlock()
	return append([]T(nil), s.s...)
}

func (s *Slice[T]) Len() int {
	s.mu.Lock()
	defer s.mu.Unlock()
	return len(s.s)
}
```

```go
package csync

import "sync"

// Map is a guarded generic map (sync.Map but typed).
type Map[K comparable, V any] struct {
	mu sync.RWMutex
	m  map[K]V
}

func (m *Map[K, V]) Store(k K, v V) {
	m.mu.Lock()
	if m.m == nil {
		m.m = make(map[K]V)
	}
	m.m[k] = v
	m.mu.Unlock()
}

func (m *Map[K, V]) Load(k K) (V, bool) {
	m.mu.RLock()
	defer m.mu.RUnlock()
	v, ok := m.m[k]
	return v, ok
}

func (m *Map[K, V]) Delete(k K) {
	m.mu.Lock()
	delete(m.m, k)
	m.mu.Unlock()
}

func (m *Map[K, V]) Range(fn func(k K, v V) bool) {
	m.mu.RLock()
	defer m.mu.RUnlock()
	for k, v := range m.m {
		if !fn(k, v) {
			break
		}
	}
}
```

```go
package tui

import (
	"context"

	"marshal/internal/pubsub"
	tea "charm.land/bubbletea/v2"
)

// pumpJobEvents subscribes to the job broker and returns a tea.Cmd that
// blocks until the next JobEvent arrives (or ctx is cancelled), then
// returns a jobCountMsg. The model calls this repeatedly from Update to
// keep the pump alive — one in-flight subscription at a time, not one
// goroutine per event (F19 R2: "one goroutine per program, not per event
// type").
func pumpJobEvents(ctx context.Context, b *pubsub.Broker[JobEvent]) tea.Cmd {
	ch := b.Subscribe(ctx)
	return func() tea.Msg {
		ev, ok := <-ch
		if !ok {
			return nil // broker closed / ctx cancelled → pump stops
		}
		return jobCountMsg{count: ev.Payload.Count}
	}
}
```

```go
// PushSteering appends a message typed by the user mid-turn. It is a
// steering message (consumed at the next loop-top) if a turn is running,
// or the next follow-up turn if the turn has already ended.
func (s *State) PushSteering(msg string) {
	s.mu.Lock()
	s.steeringQueue = append(s.steeringQueue, msg)
	broker := s.steeringBroker
	s.mu.Unlock()
	if broker != nil {
		broker.Publish("steering", SteeringEvent{QueueLen: len(s.steeringQueue), Message: msg})
	}
}

// SteeringQueue returns a copy of the current queue (does not drain).
func (s *State) SteeringQueue() []string {
	s.mu.Lock()
	defer s.mu.Unlock()
	return append([]string(nil), s.steeringQueue...)
}

// DrainSteering returns and clears the queue atomically.
func (s *State) DrainSteering() []string {
	s.mu.Lock()
	defer s.mu.Unlock()
	out := s.steeringQueue
	s.steeringQueue = nil
	return out
}

// ClearSteering drops all queued messages without returning them.
func (s *State) ClearSteering() {
	s.mu.Lock()
	s.steeringQueue = nil
	s.mu.Unlock()
}

// SetSteeringBroker wires the pub/sub broker so PushSteering publishes.
func (s *State) SetSteeringBroker(b *pubsub.Broker[SteeringEvent]) {
	s.mu.Lock()
	s.steeringBroker = b
	s.mu.Unlock()
}
```

```go
// Package diffview renders unified diffs as styled, syntax-highlighted
// side-by-side or unified text for the TUI (F17). The diff algorithm for
// intraline emphasis uses github.com/sergi/go-diff (MIT); syntax highlighting
// uses github.com/alecthomas/chroma/v2 (MIT, already in the dep tree via
// glamour). This is a clean-room implementation of the public behavior
// described in docs/12 F17; it does not derive from crush's FSL-licensed
// diffview source.
package diffview

import (
	"bufio"
	"fmt"
	"strings"
)

// Mode selects the render layout.
type Mode int

const (
	ModeAuto       Mode = iota // side-by-side when width >= 120, else unified
	ModeSideBySide             // force side-by-side
	ModeUnified                // force unified
)

// Options configures a render.
type Options struct {
	Width     int  // available terminal width
	Mode      Mode
	Highlight bool // apply chroma syntax highlighting
	Theme     *chromaStyle // nil → default theme (F17 R4: no hardcoded colors)
}

// LineKind classifies a diff line.
type LineKind int

const (
	LineContext LineKind = iota
	LineAdded
	LineRemoved
	LineHunkHeader
	LineFileHeader
)

// Line is a single parsed diff line.
type Line struct {
	Kind     LineKind
	Content  string
	OldNumber int // 0 if not an old-side line
	NewNumber int // 0 if not a new-side line
}

// Hunk is a parsed @@ hunk.
type Hunk struct {
	OldStart int
	NewStart int
	Lines    []Line
}

// Render produces a styled string for the TUI. It never returns an error —
// on parse failure it falls back to rendering the raw input as plain text
// (F17 R1: "falls back to plain text below a width floor or when
// highlighting fails").
func Render(diff string, opts Options) string {
	hunks, err := parseUnifiedDiff(diff)
	if err != nil {
		return plainTextFallback(diff, opts.Width)
	}
	mode := opts.Mode
	if mode == ModeAuto {
		if opts.Width >= 120 {
			mode = ModeSideBySide
		} else {
			mode = ModeUnified
		}
	}
	var b strings.Builder
	for i, h := range hunks {
		if i > 0 {
			b.WriteString("\n")
		}
		if mode == ModeSideBySide {
			renderSideBySide(&b, h, opts)
		} else {
			renderUnified(&b, h, opts)
		}
	}
	return b.String()
}

func parseUnifiedDiff(diff string) ([]Hunk, error) {
	var hunks []Hunk
	var cur *Hunk
	oldLine, newLine := 0, 0
	scanner := bufio.NewScanner(strings.NewReader(diff))
	scanner.Buffer(make([]byte, 0, 64*1024), 1024*1024)
	for scanner.Scan() {
		line := scanner.Text()
		switch {
		case strings.HasPrefix(line, "--- ") || strings.HasPrefix(line, "+++ "):
			if cur != nil {
				hunks = append(hunks, *cur)
				cur = nil
			}
			// file header line — tracked but not rendered as a hunk line
			continue
		case strings.HasPrefix(line, "@@ "):
			if cur != nil {
				hunks = append(hunks, *cur)
			}
			os, ns, ok := parseHunkHeader(line)
			if !ok {
				return nil, fmt.Errorf("bad hunk header: %q", line)
			}
			cur = &Hunk{OldStart: os, NewStart: ns}
			oldLine, newLine = os, ns
			cur.Lines = append(cur.Lines, Line{Kind: LineHunkHeader, Content: line})
			continue
		case strings.HasPrefix(line, "diff ") || strings.HasPrefix(line, "index "):
			if cur != nil {
				hunks = append(hunks, *cur)
				cur = nil
			}
			continue
		}
		if cur == nil {
			return nil, fmt.Errorf("diff line outside hunk: %q", line)
		}
		switch {
		case strings.HasPrefix(line, "+"):
			cur.Lines = append(cur.Lines, Line{Kind: LineAdded, Content: line[1:], NewNumber: newLine})
			newLine++
		case strings.HasPrefix(line, "-"):
			cur.Lines = append(cur.Lines, Line{Kind: LineRemoved, Content: line[1:], OldNumber: oldLine})
			oldLine++
		case strings.HasPrefix(line, " "):
			cur.Lines = append(cur.Lines, Line{Kind: LineContext, Content: line[1:], OldNumber: oldLine, NewNumber: newLine})
			oldLine++
			newLine++
		default:
			return nil, fmt.Errorf("unrecognized diff line: %q", line)
		}
	}
	if err := scanner.Err(); err != nil {
		return nil, err
	}
	if cur != nil {
		hunks = append(hunks, *cur)
	}
	if len(hunks) == 0 {
		return nil, fmt.Errorf("no hunks parsed")
	}
	return hunks, nil
}

func parseHunkHeader(line string) (oldStart, newStart int, ok bool) {
	// @@ -1,3 +1,4 @@
	rest := strings.TrimPrefix(line, "@@ ")
	parts := strings.SplitN(rest, " ", 2)
	if len(parts) < 2 {
		return 0, 0, false
	}
	oldStart, ok = parseRangeStart(parts[0])
	if !ok {
		return 0, 0, false
	}
	newStart, ok = parseRangeStart(parts[1])
	return oldStart, newStart, ok
}

func parseRangeStart(s string) (int, bool) {
	s = strings.TrimPrefix(s, "-")
	s = strings.TrimPrefix(s, "+")
	if idx := strings.IndexByte(s, ','); idx >= 0 {
		s = s[:idx]
	}
	var n int
	if _, err := fmt.Sscanf(s, "%d", &n); err != nil {
		return 0, false
	}
	return n, true
}

func plainTextFallback(diff string, width int) string {
	if width <= 0 {
		return diff
	}
	var b strings.Builder
	for _, line := range strings.Split(diff, "\n") {
		if len(line) > width {
			line = line[:width]
		}
		b.WriteString(line)
		b.WriteString("\n")
	}
	return b.String()
}
```

```go
func renderSideBySide(b *strings.Builder, h Hunk, opts Options) {
	half := (opts.Width - 3) / 2 // 3 for the " | " separator
	if half < 20 {
		renderUnified(b, h, opts) // too narrow, fall back
		return
	}
	pairs := pairLines(h.Lines)
	for _, p := range pairs {
		left, right := p.left, p.right
		lstr := renderSideColumn(left, half, removedStyle, remEmphStyle, opts, h)
		rstr := renderSideColumn(right, half, addedStyle, addEmphStyle, opts, h)
		b.WriteString(lstr)
		b.WriteString(hunkStyle.Render(" | "))
		b.WriteString(rstr)
		b.WriteString("\n")
	}
}

type sidePair struct {
	left, right *Line
}

func pairLines(lines []Line) []sidePair {
	var pairs []sidePair
	i := 0
	for i < len(lines) {
		ln := lines[i]
		if ln.Kind == LineHunkHeader || ln.Kind == LineFileHeader {
			i++
			continue
		}
		if ln.Kind == LineRemoved {
			// pair with the next added line if present
			if i+1 < len(lines) && lines[i+1].Kind == LineAdded {
				pairs = append(pairs, sidePair{left: &lines[i], right: &lines[i+1]})
				i += 2
				continue
			}
			l := lines[i]
			pairs = append(pairs, sidePair{left: &l})
			i++
			continue
		}
		if ln.Kind == LineAdded {
			r := ln
			pairs = append(pairs, sidePair{right: &r})
			i++
			continue
		}
		// context
		c := ln
		pairs = append(pairs, sidePair{left: &c, right: &c})
		i++
	}
	return pairs
}

func renderSideColumn(ln *Line, width int, baseStyle, emphStyle lipgloss.Style, opts Options, h Hunk) string {
	if ln == nil {
		return strings.Repeat(" ", width)
	}
	content := ln.Content
	if opts.Highlight {
		content = highlight(content, detectLanguage(h), baseStyle)
	}
	content = truncateVisible(content, width)
	return baseStyle.Render(padRight(content, width))
}

func padRight(s string, width int) string {
	if lipgloss.Width(s) >= width {
		return s
	}
	return s + strings.Repeat(" ", width-lipgloss.Width(s))
}

func truncateVisible(s string, width int) string {
	if lipgloss.Width(s) <= width {
		return s
	}
	// crude truncation by rune count — good enough for the width floor
	r := []rune(s)
	if len(r) > width {
		r = r[:width]
	}
	return string(r)
}
```

```go
package diffview

import (
	"strings"

	"github.com/alecthomas/chroma/v2"
	"github.com/alecthomas/chroma/v2/lexers"
	"github.com/alecthomas/chroma/v2/styles"
	"github.com/charmbracelet/lipgloss/v2"
)

// chromaStyle wraps a chroma style for lipgloss rendering. nil uses the
// default theme (F17 R4: respect the active theme; for now the default
// chroma style — a future task can map lipgloss theme colors here).
type chromaStyle = chroma.Style

var defaultChromaStyle = styles.Get("swapoff")

// detectLanguage guesses a chroma lexer from the hunk's file header.
func detectLanguage(h Hunk) string {
	// The file header lines were consumed in parseUnifiedDiff; we rely on
	// the first non-empty context line's extension if available, else "go"
	// as the Marshal default. A richer version threads the file path from
	// the parser; for now, default to Go (Marshal's primary language).
	return "go"
}

// highlight renders a code line with chroma syntax tokens, wrapped so the
// base style's foreground applies to unclassified text. Returns a string
// of lipgloss-styled fragments. On any error or missing lexer, returns the
// plain content under baseStyle (F17 R1 fallback).
func highlight(content, language string, baseStyle lipgloss.Style) string {
	lexer := lexers.Get(language)
	if lexer == nil {
		lexer = lexers.Fallback
	}
	iterator, err := lexer.Tokenise(nil, content)
	if err != nil {
		return content
	}
	var b strings.Builder
	for token := iterator(); token != chroma.EOF; token = iterator() {
		txt := token.Value
		switch token.Type {
		case chroma.Keyword:
			b.WriteString(lipgloss.NewStyle().Foreground(lipgloss.Color("5")).Render(txt))
		case chroma.String:
			b.WriteString(lipgloss.NewStyle().Foreground(lipgloss.Color("3")).Render(txt))
		case chroma.Comment:
			b.WriteString(lipgloss.NewStyle().Foreground(lipgloss.Color("8")).Render(txt))
		case chroma.Number:
			b.WriteString(lipgloss.NewStyle().Foreground(lipgloss.Color("6")).Render(txt))
		default:
			b.WriteString(txt)
		}
	}
	return b.String()
}
```

```go
package tui

import "strings"

// fuzzyScore returns a subsequence-match score for query against target
// (higher is better). A non-match returns (0, false). The scorer rewards
// consecutive matches and prefix matches so "/pl" ranks "/plan" above
// "/help". This is a simple in-package scorer (F18 design sketch: "a
// simple subsequence scorer") — no external fuzzy dep.
func fuzzyScore(query, target string) (int, bool) {
	q := strings.ToLower(query)
	t := strings.ToLower(target)
	if len(q) > len(t) {
		return 0, false
	}
	score := 0
	ti := 0
	consecutive := 0
	for qi := 0; qi < len(q); qi++ {
		matched := false
		for ; ti < len(t); ti++ {
			if q[qi] == t[ti] {
				matched = true
				score += 1 + consecutive
				consecutive++
				if ti == 0 || qi == 0 {
					score += 2 // prefix bonus
				}
				ti++
				break
			}
			consecutive = 0
		}
		if !matched {
			return 0, false
		}
	}
	return score, true
}
```

```go
type completionKind int

const (
	completionCommand completionKind = iota
	completionFile
)

type completionItem struct {
	Text        string
	Description string
	Kind        completionKind
}

type completionPopup struct {
	items        []completionItem
	filtered     []completionItem
	index        int
	visible      bool
	acceptedText string
}

func newCompletionPopup(items []completionItem) *completionPopup {
	return &completionPopup{items: items}
}

func (p *completionPopup) update(query string) {
	if query == "" {
		p.visible = false
		p.filtered = nil
		return
	}
	type scored struct {
		item  completionItem
		score int
	}
	var hits []scored
	for _, it := range p.items {
		s, ok := fuzzyScore(query, it.Text)
		if ok {
			hits = append(hits, scored{it, s})
		}
	}
	// sort by score desc, then text
	for i := 0; i < len(hits); i++ {
		for j := i + 1; j < len(hits); j++ {
			if hits[j].score > hits[i].score || (hits[j].score == hits[i].score && hits[j].item.Text < hits[i].item.Text) {
				hits[i], hits[j] = hits[j], hits[i]
			}
		}
	}
	p.filtered = make([]completionItem, len(hits))
	for i, h := range hits {
		p.filtered[i] = h.item
	}
	p.index = 0
	p.visible = len(p.filtered) > 0
}

func (p *completionPopup) matches() []completionItem { return p.filtered }
func (p *completionPopup) visible_() bool             { return p.visible }
func (p *completionPopup) dismiss()                   { p.visible = false; p.filtered = nil; p.index = 0 }

func (p *completionPopup) moveDown() {
	if len(p.filtered) == 0 {
		return
	}
	p.index = (p.index + 1) % len(p.filtered)
}
func (p *completionPopup) moveUp() {
	if len(p.filtered) == 0 {
		return
	}
	p.index = (p.index - 1 + len(p.filtered)) % len(p.filtered)
}
func (p *completionPopup) accept() {
	if len(p.filtered) == 0 {
		return
	}
	chosen := p.filtered[p.index]
	switch chosen.Kind {
	case completionCommand:
		p.acceptedText = chosen.Text + " "
	case completionFile:
		p.acceptedText = chosen.Text
	}
	p.visible = false
	p.filtered = nil
	p.index = 0
}
```

```go
// PinFiles adds the given snippets as pinned sections that bypass the
// token-budget gate (F18 R2: accepted @file references are injected as
// context, budget permitting — but pinning means they are not dropped by
// the greedy rebudget pass). Call this after Build/Rebudget so pinned
// sections are appended last and never pruned.
func PinFiles(pack Pack, snippets []FileSnippet) Pack {
	for _, snip := range snippets {
		pack.Sections = append(pack.Sections, Section{
			Kind:           SectionFileSnippet,
			Title:          snip.Path,
			Content:        snip.Content,
			Source:         snip.Path,
			Priority:       100, // higher than normal snippets
			EstimatedTokens: estimateTokens(snip.Content),
		})
	}
	pack.Pinned = append(pack.Pinned, snippets...)
	return pack
}
```

```go
var atFileRe = regexp.MustCompile(`(?:^|\s)@(\S+)`)

func extractPinnedFiles(goal string, state *session.State, projectID int64) []contextpack.FileSnippet {
	matches := atFileRe.FindAllStringSubmatch(goal, -1)
	if len(matches) == 0 {
		return nil
	}
	// Resolve each against the repo index. The runner has access to the DB
	// via state.DB(); the file index is db.GetFileIndex(projectID).
	// Build a path set for O(1) lookup.
	index, err := state.DB().GetFileIndex(projectID)
	if err != nil {
		return nil
	}
	known := make(map[string]bool, len(index))
	for _, f := range index {
		known[f.Path] = true
	}
	var out []contextpack.FileSnippet
	for _, m := range matches {
		path := m[1]
		if !known[path] {
			continue // ignore unknown references rather than erroring
		}
		content, err := os.ReadFile(filepath.Join(state.WorkingDir, path))
		if err != nil {
			continue
		}
		out = append(out, contextpack.FileSnippet{Path: path, Content: string(content)})
	}
	return out
}
```
